cmake_minimum_required(VERSION 3.20)
project(SlicePilotAssistantUITest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SLICEPILOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(APPLE AND NOT wxWidgets_CONFIG_EXECUTABLE)
    set(wxWidgets_CONFIG_EXECUTABLE "${SLICEPILOT_ROOT}/deps/build/${CMAKE_SYSTEM_PROCESSOR}/OrcaSlicer_dep/usr/local/bin/wx-config")
endif()
find_package(wxWidgets REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
add_executable(slicepilot_ui_test ui_smoke.cpp "${SLICEPILOT_ROOT}/src/slic3r/GUI/CodexAssistant.cpp")
target_include_directories(slicepilot_ui_test PRIVATE "${SLICEPILOT_ROOT}/src" "${SLICEPILOT_ROOT}/deps_src")
# Match the upstream dependency build (wxBUILD_DEBUG_LEVEL=0).
target_compile_definitions(slicepilot_ui_test PRIVATE wxDEBUG_LEVEL=0)
target_link_libraries(slicepilot_ui_test PRIVATE ${wxWidgets_LIBRARIES})
enable_testing()
add_test(NAME slicepilot_native_conversation COMMAND slicepilot_ui_test --self-test)
set_tests_properties(slicepilot_native_conversation PROPERTIES TIMEOUT 30)

add_test(NAME slicepilot_rejects_untrusted_login COMMAND slicepilot_ui_test --self-test-invalid-login)
set_tests_properties(slicepilot_rejects_untrusted_login PROPERTIES TIMEOUT 30)
